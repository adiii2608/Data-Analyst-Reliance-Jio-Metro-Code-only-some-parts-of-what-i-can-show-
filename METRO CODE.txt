// Constants
val base32 = "0123456789bcdefghjkmnpqrstuvwxyz"
val earthRadiusKm = 6371.0

// Function to convert degrees to radians
def degToRad(deg: Double): Double = deg * Math.PI / 180.0

// Function to   radians to degrees
def radToDeg(rad: Double): Double = rad * 180.0 / Math.PI

// Haversine formula to calculate the distance between two points on the Earth's surface
def haversine(lat1: Double, lon1: Double, lat2: Double, lon2: Double): Double = {
  val dLat = degToRad(lat2 - lat1)
  val dLon = degToRad(lon2 - lon1)
  val a = Math.sin(dLat / 2) * Math.sin(dLat / 2) +
    Math.cos(degToRad(lat1)) * Math.cos(degToRad(lat2)) *
    Math.sin(dLon / 2) * Math.sin(dLon / 2)
  val c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
  earthRadiusKm * c
}

// Function to calculate new latitude and longitude given a distance and bearing
def calculateNewCoordinates(lat: Double, lon: Double, distanceKm: Double, bearingDegrees: Double): (Double, Double) = {
  val bearing = degToRad(bearingDegrees)
  val latRad = degToRad(lat)
  val lonRad = degToRad(lon)

  val newLatRad = Math.asin(Math.sin(latRad) * Math.cos(distanceKm / earthRadiusKm) +
    Math.cos(latRad) * Math.sin(distanceKm / earthRadiusKm) * Math.cos(bearing))
  val newLonRad = lonRad + Math.atan2(Math.sin(bearing) * Math.sin(distanceKm / earthRadiusKm) * Math.cos(latRad),
    Math.cos(distanceKm / earthRadiusKm) - Math.sin(latRad) * Math.sin(newLatRad))

  (radToDeg(newLatRad), radToDeg(newLonRad))
}

// Function to encode a latitude and longitude into a geohash
def encodeGeohash(latitude: Double, longitude: Double, precision: Int): String = {
  var latMin = -90.0
  var latMax = 90.0
  var lonMin = -180.0
  var lonMax = 180.0
  val geohash = new StringBuilder()
  var isEven = true
  var bit = 0
  var ch = 0

  while (geohash.length < precision) {
    if (isEven) {
      val mid = (lonMin + lonMax) / 2
      if (longitude > mid) {
        ch |= (1 << (4 - bit))
        lonMin = mid
      } else {
        lonMax = mid
      }
    } else {
      val mid = (latMin + latMax) / 2
      if (latitude > mid) {
        ch |= (1 << (4 - bit))
        latMin = mid
      } else {
        latMax = mid
      }
    }
    isEven = !isEven
    if (bit < 4) {
      bit += 1
    } else {
      geohash.append(base32.charAt(ch))
      bit = 0
      ch = 0
    }
  }
  geohash.toString()
}


// Function to generate geohashes within a bounding box and filter by distance
def geohashesInBoundingBox(lat: Double, lon: Double, precision: Int, distanceKm: Double): List[String] = {
  val northWest = calculateNewCoordinates(lat, lon, distanceKm, 315) // NW
  val southEast = calculateNewCoordinates(lat, lon, distanceKm, 135) // SE

  val latMin = southEast._1
  val latMax = northWest._1
  val lonMin = northWest._2
  val lonMax = southEast._2

  val geohashes = ListBuffer[String]()

  val stepSize = 0.0001 // Adjust step size for better coverage

  var latVal = latMin
  while (latVal <= latMax) {
    var lonVal = lonMin
    while (lonVal <= lonMax) {
      val geohash = encodeGeohash(latVal, lonVal, precision)
      if (haversine(lat, lon, latVal, lonVal) <= distanceKm) {
        geohashes += geohash
      }
      lonVal += stepSize
    }
    latVal += stepSize
  }

  geohashes.toList.distinct
}

val routine_features ={
  dominant_grids
     .withColumn("imsi", concat(lit("`"),col("imsi").cast("string")))
    .groupBy("imsi")
    .pivot("time_window")
    .agg(
      first("geohashid").alias("grid"),
      first("days_present").alias("days")
    )
    .withColumnRenamed("home_grid","home_geohash")
    .withColumnRenamed("office_grid","office_geohash")
    .withColumnRenamed("home_days","home_days")
    .withColumnRenamed("office_days","office_days")
    .withColumn(
      "routine_strength",
      col("office_days") / (col("home_days") + col("office_days"))
    )
    .withColumn(
      "regular_commuter_flag",
      when(col("office_days") >= 3, 1).otherwise(0)
    )
    .filter($"home_days" >= 2)

}
